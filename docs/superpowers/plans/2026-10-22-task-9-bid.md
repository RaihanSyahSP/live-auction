### PlaceBid

```go
func (s *bidService) PlaceBid(
	ctx context.Context,
	userID uuid.UUID,
	auctionID uuid.UUID,
	req bid_request.PlaceBidRequest,
) (bid_response.PlaceBidResponse, error) {
	const maxAmount int64 = 1_000_000_000_000

	if req.Amount <= 0 || req.Amount > maxAmount {
		return bid_response.PlaceBidResponse{}, exception.ErrInvalidAmount
	}

	var result bid_response.PlaceBidResponse

	err := s.tx.WithinTx(ctx, func(tx pgx.Tx) error {
		auction, err := s.auctions.LockByID(ctx, tx, auctionID)
		if err != nil {
			return err
		}

		now := s.now().UTC()

		// Replay harus dicek sebelum validasi harga.
		existing, err := s.bids.FindByIdempotencyKey(
			ctx,
			tx,
			auctionID,
			userID,
			req.IdempotencyKey,
		)
		if err == nil {
			if existing.Amount != req.Amount {
				return exception.ErrIdempotencyConflict
			}

			result = bid_response.PlaceBidResponse{
				Auction: toAuctionResponse(auction),
				Bid:     toBidResponse(existing),
				Replay:  true,
			}
			return nil
		}
		if !errors.Is(err, exception.ErrBidNotFound) {
			return err
		}

		if auction.Status != domain.AuctionStatusLive {
			return exception.ErrAuctionNotLive
		}
		if auction.EndsAt == nil || !now.Before(*auction.EndsAt) {
			return exception.ErrAuctionNotLive
		}
		if auction.CreatedBy == userID {
			return exception.ErrCreatorCannotBid
		}
		if auction.HighestBidderID != nil && *auction.HighestBidderID == userID {
			return exception.ErrAlreadyHighest
		}

		minimum := auctionrules.MinimumBid(auction)
		if req.Amount < minimum {
			return exception.ErrBidTooLow
		}

		previousHighest := auction.HighestBidderID
		extended := false

		newEndsAt, shouldExtend := auctionrules.NextEndsAt(
			now,
			*auction.EndsAt,
			auction.ExtensionCount,
		)
		if shouldExtend {
			auction.EndsAt = &newEndsAt
			auction.ExtensionCount++
			extended = true
		}

		auction.CurrentPrice = req.Amount
		auction.HighestBidderID = &userID
		auction.Version++

		inserted, err := s.bids.Insert(ctx, tx, domain.Bid{
			AuctionID:      auctionID,
			UserID:         userID,
			Amount:         req.Amount,
			IdempotencyKey: req.IdempotencyKey,
		})
		if errors.Is(err, exception.ErrBidAlreadyExists) {
			// Row lock biasanya sudah mencegah ini. Ini hanya safety net.
			return exception.ErrIdempotencyConflict
		}
		if err != nil {
			return err
		}

		updated, err := s.auctions.Update(ctx, tx, auction)
		if err != nil {
			return err
		}

		result = bid_response.PlaceBidResponse{
			Auction:               toAuctionResponse(updated),
			Bid:                   toBidResponse(inserted),
			Replay:                false,
			Extended:              extended,
			PreviousHighestUserID: previousHighest,
		}
		return nil
	})

	return result, err
}
```

## 5. Bid controller

Buat `internal/controller/bid_controller.go`:

```go
package controller

import (
	"net/http"

	"github.com/google/uuid"

	"live-auction/server/internal/exception"
	"live-auction/server/internal/helper"
	"live-auction/server/internal/middleware"
	"live-auction/server/internal/model/web/bid_request"
	"live-auction/server/internal/service"
)

type BidController struct {
	bidService service.BidService
}

func NewBidController(bidService service.BidService) *BidController {
	return &BidController{bidService: bidService}
}

func (c *BidController) PlaceBid(w http.ResponseWriter, r *http.Request) {
	principal, ok := middleware.PrincipalFrom(r.Context())
	if !ok {
		exception.WriteError(w, exception.ErrUnauthenticated)
		return
	}

	auctionID, err := uuid.Parse(r.PathValue("id"))
	if err != nil {
		exception.WriteError(w, exception.ErrValidation)
		return
	}

	var req bid_request.PlaceBidRequest
	if err := helper.ReadJSON(w, r, &req); err != nil {
		exception.WriteError(w, exception.ErrValidation)
		return
	}

	result, err := c.bidService.PlaceBid(
		r.Context(),
		principal.UserID,
		auctionID,
		req,
	)
	if err != nil {
		exception.WriteError(w, err)
		return
	}

	status := http.StatusCreated
	if result.Replay {
		status = http.StatusOK
	}

	helper.WriteJSON(w, status, result)
}
```

## 6. Rate limit per user

Tambahkan helper di `internal/middleware/rate_limit.go`:

```go
func UserID(r *http.Request) string {
	principal, ok := PrincipalFrom(r.Context())
	if !ok {
		return ClientIP(r)
	}
	return principal.UserID.String()
}
```

Rate limit bid: 3 per detik, burst 5.

## 7. Router

Tambahkan field:

```go
Bid        *controller.BidController
BidLimiter *middleware.RateLimiter
```

Daftarkan route:

```go
if deps.Bid != nil && deps.AccessTokens != nil {
	bidHandler := middleware.RateLimit(
		deps.BidLimiter,
		middleware.UserID,
		http.HandlerFunc(deps.Bid.PlaceBid),
	)
	mux.Handle(
		"POST /api/auctions/{id}/bids",
		middleware.Origin(
			deps.AllowedOrigin,
			middleware.Auth(deps.AccessTokens, bidHandler),
		),
	)
}
```

Urutannya penting:

```text
Origin -> Auth -> RateLimit -> BidController
```

- Origin menolak request lintas origin.
- Auth mengisi principal.
- RateLimit memakai `UserID` dari principal.
- Controller memanggil service.

Kalau RateLimit diletakkan sebelum Auth, `UserID` tidak tersedia dan rate limit akan jatuh ke IP.

## 9. Test service

Gunakan repository asli dan transaksi yang di-rollback seperti test auction service.

Kasus minimum:

- bid pertama sebesar `starting_price` berhasil
- bid di bawah minimum menghasilkan `ErrBidTooLow`
- pembuat lelang bid menghasilkan `ErrCreatorCannotBid`
- bidder yang sudah highest menghasilkan `ErrAlreadyHighest`
- bid setelah `ends_at` menghasilkan `ErrAuctionNotLive`
- bid saat status draft atau ended menghasilkan `ErrAuctionNotLive`
- bid dalam 60 detik terakhir memperpanjang `ends_at` dan menaikkan `extension_count`
- extension berhenti setelah 5 kali
- bid sukses menaikkan `current_price`, `highest_bidder_id`, dan `version`
- idempotency key sama dengan amount sama menghasilkan `Replay=true`
- idempotency key sama dengan amount berbeda menghasilkan `ErrIdempotencyConflict`
- dua bidder bersamaan pada nominal sama: satu sukses, satu `ErrBidTooLow`
- bid pada auction yang tidak ada menghasilkan `ErrAuctionNotFound`

Contoh test anti-sniping:
